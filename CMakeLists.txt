cmake_minimum_required(VERSION 3.10)
project(LinuxUserManager VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_executable(policy-tests tests/policy_tests.cpp)
add_test(NAME account-policy COMMAND policy-tests)
option(POLICY_ONLY "Build policy tests without Qt" OFF)
if(NOT POLICY_ONLY)
  find_package(Qt5 5.12 REQUIRED COMPONENTS Widgets)
  add_executable(linux-user-manager src/main.cpp)
  target_link_libraries(linux-user-manager PRIVATE Qt5::Widgets)
  install(TARGETS linux-user-manager RUNTIME DESTINATION bin)
endif()
